# v1.3-html — Browser CBA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a simplified 9-stream cost-benefit report to the browser demo (`web/index.html`), mirroring the Python `sir.cba` framework with mean-field simplifications and per-capita normalization.

**Architecture:** Single HTML file. Extends the existing Pyodide `simulate()` to compute 9 cost streams alongside disease/healthcare arrays. JavaScript renders a new CBA card with stacked-bar + delta table; outcome dropdown gains 3 new entries; summary card grid gains 2 new cards. Per the design at `docs/superpowers/specs/2026-05-15-v1.3-html-cba-design.md`.

**Tech Stack:** Existing — Pyodide (numpy), Plotly, MathJax, vanilla JS. No new dependencies.

- [ ] **Step 2: Compute the 9 CBA streams after the existing daily loop**

Locate the section in `simulate()` where mortality outputs are computed (look for `cum_deaths = np.cumsum(daily_deaths)`). After the mortality block and BEFORE the `return { ... }` dict, append:

```python
    # ============================ CBA: 9 cost streams ============================
    # Hidden constants (Tier C — not exposed in UI)
    DW_SYMP = 0.15
    DW_HOSP = 0.50
    DW_ICU = 0.80
    WORKING_DAYS_PER_YEAR = 260
    REMAINING_WORKING_YEARS = 20  # mean-field; no age structure in this model

    # Daily discount factor array
    days = np.arange(T + 1)
    disc = (1.0 + cba_rate) ** (-days / 365.0)

    # Effective sick days per case (with mean-field hospitalization & ICU rates)
    sick_days_symptomatic = 7.0
    sick_days_hosp_extra = 14.0
    sick_days_icu_extra = 21.0
    sick_days_effective = (
        sick_days_symptomatic
        + hosp_rate * sick_days_hosp_extra
        + icu_rate * sick_days_icu_extra
    )

    # Stream 1: Direct medical
    s1_raw = hosp_prev * hcost + icu_prev * icucost + new_inf_h * outpcost
    s1_per_day = s1_raw * disc / N
    s1_total = float(s1_per_day.sum())

    # Stream 2: Vaccination program
    per_dose_cost = dosecost + admincost + sidefx
    s2_raw = new_vax_h * per_dose_cost
    s2_per_day = s2_raw * disc / N
    s2_total = float(s2_per_day.sum())

    # Stream 3: Productivity from illness
    s3_raw = new_inf_h * wage * sick_days_effective
    s3_per_day = s3_raw * disc / N
    s3_total = float(s3_per_day.sum())

    # Stream 4: Productivity from death
    loss_per_death = wage * REMAINING_WORKING_YEARS * WORKING_DAYS_PER_YEAR
    s4_raw = daily_deaths * loss_per_death
    s4_per_day = s4_raw * disc / N
    s4_total = float(s4_per_day.sum())

    # Stream 5: YLL with PV-form discounting
    if cba_rate > 0:
        pv_yll = (1.0 - np.exp(-cba_rate * yll_per_death)) / cba_rate
    else:
        pv_yll = yll_per_death
    s5_raw = daily_deaths * pv_yll
    s5_per_day = s5_raw * disc / N
    s5_total = float(s5_per_day.sum())

    # Stream 6: Acute morbidity QALY (mean-field)
    qaly_per_case = (
        DW_SYMP * sick_days_symptomatic / 365.0
        + hosp_rate * DW_HOSP * sick_days_hosp_extra / 365.0
        + icu_rate * DW_ICU * sick_days_icu_extra / 365.0
    )
    s6_raw = new_inf_h * qaly_per_case
    s6_per_day = s6_raw * disc / N
    s6_total = float(s6_per_day.sum())

    # Stream 7: Behavioral utility loss (HTML has full theta history)
    lost_meetings = (1.0 - theta_h) * m0
    s7_raw = lost_meetings * hpermeeting * vot * N  # N to undo per-capita later
    s7_per_day = s7_raw * disc / N
    s7_total = float(s7_per_day.sum())

    # Stream 8: Precaution effort cost (HTML has full e history; e is per-agent so x N)
    s8_raw = (e_h ** 2) * vot * hpermeeting * N
    s8_per_day = s8_raw * disc / N
    s8_total = float(s8_per_day.sum())

    # Stream 9: Policy costs (caller supplies constant daily $; per-capita)
    s9_raw = np.full(T + 1, policy_cost_per_day, dtype=np.float64)
    s9_per_day = s9_raw * disc / N
    s9_total = float(s9_per_day.sum())

    # Aggregates
    monetary_streams = [s1_total, s2_total, s3_total, s4_total, s7_total, s8_total, s9_total]
    total_monetary_per_capita = float(sum(monetary_streams))
    total_yll_per_capita = s5_total
    total_qaly_per_capita = s6_total

    cba = {
        "streams": {
            "direct_medical": s1_total,
            "vaccination_program": s2_total,
            "productivity_illness": s3_total,
            "productivity_death": s4_total,
            "yll": s5_total,
            "morbidity_qaly": s6_total,
            "behavioral_loss": s7_total,
            "precaution_cost": s8_total,
            "policy_costs": s9_total,
        },
        "units": {
            "direct_medical": "$/p",
            "vaccination_program": "$/p",
            "productivity_illness": "$/p",
            "productivity_death": "$/p",
            "yll": "YLL/p",
            "morbidity_qaly": "QALY/p",
            "behavioral_loss": "$/p",
            "precaution_cost": "$/p",
            "policy_costs": "$/p",
        },
        "total_monetary_per_capita": total_monetary_per_capita,
        "total_yll_per_capita": total_yll_per_capita,
        "total_qaly_per_capita": total_qaly_per_capita,
        "per_day": {
            "direct_medical": s1_per_day.tolist(),
            "vaccination_program": s2_per_day.tolist(),
            "productivity_illness": s3_per_day.tolist(),
            "productivity_death": s4_per_day.tolist(),
            "yll": s5_per_day.tolist(),
            "morbidity_qaly": s6_per_day.tolist(),
            "behavioral_loss": s7_per_day.tolist(),
            "precaution_cost": s8_per_day.tolist(),
            "policy_costs": s9_per_day.tolist(),
        },
        "cumulative_monetary": (
            s1_per_day + s2_per_day + s3_per_day + s4_per_day
            + s7_per_day + s8_per_day + s9_per_day
        ).cumsum().tolist(),
        "cumulative_yll": s5_per_day.cumsum().tolist(),
    }
```

    const STREAM_DISPLAY_ORDER = [
      "direct_medical", "vaccination_program",
      "productivity_illness", "productivity_death",
      "behavioral_loss", "precaution_cost", "policy_costs",
      "yll", "morbidity_qaly",

    const STREAM_LABELS = {
      direct_medical: "Direct medical",
      vaccination_program: "Vaccination program",
      productivity_illness: "Productivity (illness)",
      productivity_death: "Productivity (death)",
      behavioral_loss: "Behavioral loss",
      precaution_cost: "Precaution effort",
      policy_costs: "Policy direct costs",
      yll: "YLL",
      morbidity_qaly: "Morbidity QALY",
    };

    const STREAM_COLORS = {
      direct_medical: "#0891b2",
      vaccination_program: "#7c3aed",
      productivity_illness: "#ea580c",
      productivity_death: "#9f1239",
      behavioral_loss: "#84cc16",
      precaution_cost: "#facc15",
      policy_costs: "#94a3b8",
      yll: "#be123c",
      morbidity_qaly: "#db2777",
    };

    function formatMoney(v) {
      const abs = Math.abs(v);
      if (abs >= 1e6) return '$' + (v / 1e6).toFixed(2) + 'M';
      if (abs >= 1e3) return '$' + (v / 1e3).toFixed(1) + 'k';
      return '$' + v.toFixed(0);
    }

    function formatStreamValue(v, unit) {
      if (unit === '$/p') return formatMoney(v);
      return v.toFixed(3);
    }

    function renderCBACard(data, inp) {
      if (!data.with_int.cba || !data.without_int.cba) return;
      const wcba = data.with_int.cba;
      const wocba = data.without_int.cba;
      const deltaMonetary = wcba.total_monetary_per_capita - wocba.total_monetary_per_capita;
      const yllAverted = wocba.total_yll_per_capita - wcba.total_yll_per_capita;

      // Top-level summary
      document.getElementById('cba-top-delta').textContent = formatMoney(deltaMonetary);
      document.getElementById('cba-top-yll').textContent = yllAverted.toFixed(4) + ' YLL/p';
      const ratioEl = document.getElementById('cba-top-ratio');
      if (Math.abs(yllAverted) > 1e-6) {
        ratioEl.textContent = formatMoney(deltaMonetary / yllAverted);
      } else {
        ratioEl.textContent = '—';
      }

      // Stacked-bar Plotly trace
      const monetaryStreams = STREAM_DISPLAY_ORDER.filter(
        k => wcba.units[k] === '$/p'
      );
      const traces = [];
      for (const stream of monetaryStreams) {
        traces.push({
          x: ['Baseline', 'Treatment'],
          y: [wocba.streams[stream], wcba.streams[stream]],
          name: STREAM_LABELS[stream],
          type: 'bar',
          marker: { color: STREAM_COLORS[stream] },
        });
      }
      const layout = {
        barmode: 'stack',
        margin: { l: 60, r: 20, t: 10, b: 40 },
        yaxis: { title: '$ / person', gridcolor: '#f0f0f0' },
        xaxis: { gridcolor: '#f0f0f0' },
        legend: { orientation: 'v', x: 1.02, y: 1.0, font: { size: 11 } },
        font: { family: 'Inter, sans-serif', size: 12 },
        paper_bgcolor: 'white', plot_bgcolor: 'white',
      };
      Plotly.react('plot-cba', traces, layout, { responsive: true, displayModeBar: false });

      // Stream-by-stream table
      const tbody = document.getElementById('cba-table-body');
      tbody.innerHTML = '';
      for (const stream of STREAM_DISPLAY_ORDER) {
        const unit = wcba.units[stream];
        const b = wocba.streams[stream];
        const t = wcba.streams[stream];
        const d = t - b;
        const goodIfLower = !(stream === 'vaccination_program' || stream === 'policy_costs');
        const goodClass = (d < 0 && goodIfLower) || (d > 0 && !goodIfLower)
          ? 'delta-good'
          : (Math.abs(d) < 1e-9 ? '' : 'delta-bad');
        const dStr = (d >= 0 ? '+' : '−') + formatStreamValue(Math.abs(d), unit);
        const row = `<tr>
          <td>${STREAM_LABELS[stream]}</td>
          <td>${formatStreamValue(b, unit)}</td>
          <td>${formatStreamValue(t, unit)}</td>
          <td class="${goodClass}">${dStr}</td>
          <td>${unit}</td>
        </tr>`;
        tbody.insertAdjacentHTML('beforeend', row);
      }
      // Total row
      const totalRow = `<tr class="total">
        <td>TOTAL monetary</td>
        <td>${formatMoney(wocba.total_monetary_per_capita)}</td>
        <td>${formatMoney(wcba.total_monetary_per_capita)}</td>
        <td class="${deltaMonetary < 0 ? 'delta-good' : 'delta-bad'}">${(deltaMonetary >= 0 ? '+' : '−') + formatMoney(Math.abs(deltaMonetary))}</td>
        <td>$/p</td>
      </tr>`;
      tbody.insertAdjacentHTML('beforeend', totalRow);
    }

    // Wire the expand/collapse header click
    document.getElementById('cba-card-header').addEventListener('click', () => {
      const card = document.getElementById('cba-card');
      const expanded = card.dataset.expanded === 'true';
      card.dataset.expanded = expanded ? 'false' : 'true';
      document.getElementById('cba-card-toggle').textContent = expanded ? '▸ Expand' : '▾ Collapse';
      // Re-render plot when expanding (Plotly needs visible container)
      if (!expanded && lastResult) renderCBACard(lastResult, lastInp);
    });
